
3. **导出**：仅在获准进行只读导出后，在站长本机 PowerShell 7.4+ 执行下面命令；`netcup` 必须是已核对的 SSH 主机别名。成功：两条 SSH 退出 0、文件非空、tar 可列目录、SHA-256 已记录，SQL 末尾有 `Dump completed` 且表数与现场相符。SQL 和 uploads 跨系统不保证同一时点，须在受控窗口复核。

```powershell
$key = 'E:\破甲工作区\.ssh\packrivo_ed25519'
$out = 'E:\破甲工作区\r1-production-export-' + (Get-Date -Format 'yyyyMMdd-HHmmss')
New-Item -ItemType Directory -Path $out -ErrorAction Stop | Out-Null
$sql = 'docker exec mysql sh -c ''export MYSQL_PWD="$MYSQL_ROOT_PASSWORD"; exec mysqldump -uroot --single-transaction --quick --skip-lock-tables --skip-add-locks --no-tablespaces --set-gtid-purged=OFF --routines --events --triggers --default-character-set=utf8mb4 www_packrivo_com'''
ssh -i $key netcup $sql > "$out\db.sql"
if ($LASTEXITCODE -ne 0 -or (Get-Item "$out\db.sql").Length -eq 0) { throw 'SQL 导出失败，停止' }
$uploads = 'docker exec php tar -C /var/www/html/www.packrivo.com/wordpress/wp-content -czf - uploads'
ssh -i $key netcup $uploads > "$out\uploads.tar.gz"
if ($LASTEXITCODE -ne 0 -or (Get-Item "$out\uploads.tar.gz").Length -eq 0) { throw 'uploads 导出失败，停止' }
tar -tzf "$out\uploads.tar.gz" > "$out\uploads-files.txt"
if ($LASTEXITCODE -ne 0) { throw 'uploads 校验失败，停止' }
Get-FileHash -Algorithm SHA256 "$out\db.sql", "$out\uploads.tar.gz" | Format-Table
```
